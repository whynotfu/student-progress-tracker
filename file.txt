mkdir -p reports
REF="--reference-url=jdbc:postgresql://10.10.0.9:5432/nexora --reference-username=nexora --reference-password=nexora"
TGT="--url=jdbc:postgresql://10.10.0.10:5432/nexora --username=nexora --password=nexora"
CL=reports/sync-prod-%build.number%.xml

echo "===== DIFF BEFORE (stage = reference, prod = target) ====="
liquibase $TGT $REF diff

echo "===== GENERATE CHANGELOG ====="
liquibase $TGT $REF --changelog-file=$CL diff-changelog
cat $CL

echo "===== APPLY TO PROD ====="
liquibase $TGT --changelog-file=$CL update

echo "===== DIFF AFTER (must be empty) ====="
liquibase $TGT $REF diff
